Curl downloading a script file

Captures curl activity that involves downloaded executable script files, which is unusual in any corporate environment.

Enabled: ✅ true
Severity: 🔴 High

📘 Playbook 
// Curl dowloading a script file.
let ScriptExtensions = dynamic([".bash", ".bat", ".cmd", ".hta", ".js", ".jse", ".ksh", ".p1", ".php", ".ps1", ".psd1", ".psm1", ".py", ".rb", ".sh", ".vbe", ".vbs", ".wsf", ".wsh", ".zsh"]);
let DownloadProtocols = dynamic(["ftp://", "ftps://", "http://", "https://", "sftp://"]);
union(SecurityEvent
| extend CommandLine = tolower(CommandLine), NewProcessName = tolower(NewProcessName)
| where EventID == 4688 and NewProcessName endswith @"\curl.exe" and CommandLine has_any (DownloadProtocols) and CommandLine has_any (ScriptExtensions)
| project TimeGenerated, Device = Computer, Account = SubjectUserName, ParentProcess = ParentProcessName, Process = NewProcessName, CommandLine = CommandLine, Source = "SecurityEvent"),
(DeviceProcessEvents
| extend ProcessCommandLine = tolower(ProcessCommandLine), FileName = tolower(FileName)
| where FileName =~ "curl.exe" and ProcessCommandLine has_any (DownloadProtocols) and ProcessCommandLine has_any (ScriptExtensions)
| project TimeGenerated, Device = DeviceName, Account = AccountName, ParentProcess = InitiatingProcessParentFileName, Process = FileName, CommandLine = ProcessCommandLine, Source = "DeviceProcessEvents")
| order by TimeGenerated desc

Multiple network discovery commands run in a short period of time

Detects multiple LOLBin and PowerShell commands that attackers frequently use to perform reconnaissance on a network.

Enabled: ✅ true
Severity: 🟠 Low

📘 Playbook 
// Detect multiple network discovery commands run on a single host, by a single user, within a short period of time; excludes common legitimate patterns and system account activity.
let TimeWindow = 10m;
let DiscoveryTools = dynamic(["arp.exe", "dsget.exe", "dsquery.exe", "hostname.exe", "ipconfig.exe", "nbtstat.exe", "net.exe", "net1.exe", "netsh.exe", "nltest.exe", "nslookup.exe", "ping.exe", "query.exe", "quser.exe", "qwinsta.exe", "route.exe", "systeminfo.exe", "tracert.exe", "whoami.exe", "whois.exe", "wmic.exe"]);
let PowerShellDiscoveryTerms = dynamic(["get-adcomputer", "get-addomain", "get-adforest", "get-aduser", "get-ciminstance win32_networkadapterconfiguration", "get-dnsclientserveraddress", "get-localgroup", "get-localuser", "get-netadapter", "get-netipaddress", "get-netipconfiguration", "get-netneighbor", "get-netroute", "get-nettcpconnection", "get-smbconnection", "get-smbshare", "resolve-dnsname", "test-connection"]);
let Security4688 = SecurityEvent
| where EventID == 4688
| extend ProcessName = tolower(tostring(NewProcessName)), CommandLine = tolower(tostring(CommandLine)), Account = strcat(SubjectDomainName, "\\", SubjectUserName), Device = Computer;
let MDEProcess = DeviceProcessEvents
| extend ProcessName = tolower(tostring(FileName)), CommandLine = tolower(tostring(ProcessCommandLine)), Account = tostring(InitiatingProcessAccountName), Device = DeviceName;
let AllProcesses = union isfuzzy = true Security4688, MDEProcess
| extend NormalizedProcess = tolower(case(ProcessName endswith "net1.exe", "net.exe", ProcessName)), CommandLine = tolower(CommandLine)
| project TimeGenerated, Device, Account, NormalizedProcess, CommandLine;
let DiscoveryEvents = AllProcesses
| where NormalizedProcess in (DiscoveryTools) or (NormalizedProcess in ("powershell.exe","powershell_ise.exe", "pwsh.exe") and CommandLine has_any (PowerShellDiscoveryTerms))
| extend DiscoveryTechnique = case(NormalizedProcess in ("powershell.exe","powershell_ise.exe", "pwsh.exe"), "PowerShell", NormalizedProcess);
DiscoveryEvents
| summarize DiscoveryCount = dcount(DiscoveryTechnique), ToolsUsed = array_sort_asc(make_set(DiscoveryTechnique)), Commands = make_set(CommandLine) by Device, Account, bin(TimeGenerated, TimeWindow)
| where not (set_has_element(ToolsUsed, "net.exe") and set_has_element(ToolsUsed, "ipconfig.exe") and set_has_element(ToolsUsed, "netsh.exe") and array_length(ToolsUsed) == 3) // Not necessary if DiscoveryCount set to 4 or higher.
// DiscoveryCount > 3 ensures multiple distinct discovery techniques, not repeated execution of the same tool.
| where DiscoveryCount > 3 and not (Account == "system" or Account endswith "$")
| sort by TimeGenerated desc

Suspicious Remote Access Tool activity

Detects any of several known legitimate remote access applications making successful remote port connections.

Enabled: ❌ false (Disabled)
Severity: 🟡 Medium

📘 Playbook 
let ReservedIPs = (ip:string) {ipv4_is_match(ip, "10.0.0.0/8") or ipv4_is_match(ip, "100.64.0.0/10") or ipv4_is_match(ip, "127.0.0.0/8") or ipv4_is_match(ip, "169.254.0.0/16") or ipv4_is_match(ip, "172.16.0.0/12") or ipv4_is_match(ip,"192.168.0.0/16") or ipv4_is_match(ip, "224.0.0.0/24") or ipv4_is_match(ip, "239.0.0.0/8")};
let RATs = _GetWatchlist('RATs') | project tolower(SearchKey);
let AdminRATs = _GetWatchlist('AdminRATs') | project tolower(SearchKey);
let PrivilegedAccounts = _GetWatchlist('Privileged_Accounts') | project Account = tolower(SearchKey);
DeviceNetworkEvents
| where TimeGenerated > ago(1d)
| extend InitiatingProcessAccountName = tolower(InitiatingProcessAccountName), InitiatingProcessFileName = tolower(InitiatingProcessFileName)
// Find all RAT capable applications making successful RemotePort connections.
| where ActionType == "ConnectionSuccess" and isnotempty(RemotePort) and (InitiatingProcessFileName in (RATs) or InitiatingProcessFileName in (AdminRATs))
// Exclude connections to internal destinations.
| where not(ReservedIPs(RemoteIP))
// Always include commercial RATs.
| where (InitiatingProcessFileName in (RATs)
// Exclude traffic involving commonly approved RATs, if the traffic originates from privileged accounts,
or (InitiatingProcessFileName in (AdminRATs) and not((InitiatingProcessAccountName in (PrivilegedAccounts)
// also exclude traffic that originates from Windows or computer accounts,
or InitiatingProcessAccountName in~ ("system", "local service", "network service") or InitiatingProcessAccountName endswith "$")
// and exclude new computer setups.
or (DeviceName startswith "desktop-" and InitiatingProcessFileName =~ "powershell.exe"))))
// Group events by user, device, process, and day.
| summarize Events = count(), First_Event = min(TimeGenerated), Last_Event = max(TimeGenerated), RemoteIPs = make_set(RemoteIP), DistinctIPs = dcount(RemoteIP), RemotePorts = make_set(RemotePort), DistinctPorts = dcount(RemotePort) by InitiatingProcessAccountName, DeviceName, InitiatingProcessFileName, Day = bin(TimeGenerated, 1d)

Unusual File Creation or Modification Process Chain

Detects files being created or modified by one application and then executed by another within 5 minutes. One exclusion has been added for CCMExec, but a confidence mechanism has also been added for future trigger handling.

Enabled: ✅ true
Severity: 🟠 Low

📘 Playbook 
// Unusual process creation or modification process chain: find a file that was executed or loaded within 5 minutes of being created or modified.
let lookback = ago(1h);
let endwindow = now();
// Additional executable to consider, depending on environment: msiexec.exe.
let MicrosoftBins = dynamic(["attrib.exe", "bash.exe", "bitsadmin.exe", "certreq.exe", "certutil.exe", "cmd.exe", "compact.exe", "copy.exe", "cscript.exe", "curl.exe", "desktopimgdownldr.exe", "dxcap.exe", "esentutl.exe", "expand.exe", "extrac32.exe", "findstr.exe", "forfiles.exe", "icacls.exe", "ie4uinit.exe", "installutil.exe", "makecab.exe", "mavinject.exe", "move.exe", "mshta.exe","node.exe", "perl.exe", "powershell.exe", "powershell_ise.exe", "pwsh.exe", "python.exe", "pythonw.exe", "reg.exe", "regsvr32.exe", "replace.exe", "robocopy.exe", "ruby.exe", "rundll32.exe", "schtasks.exe", "tar.exe", "wevtutil.exe", "wget.exe", "wmic.exe", "wscript.exe", "xcopy.exe"]);
let RiskyExtensions = dynamic([ ".7z", ".bash", ".bat", ".cgi", ".cmd", ".cpl", ".dat", ".dll", ".exe", ".gz", ".hta", ".htm", ".html",".img", ".iso", ".js", ".jse", ".ksh", ".mht", ".mhtml", ".msi", ".msp", ".mst", ".ocx", ".p1", ".php", ".pl", ".ps1", ".psm1", ".py", ".rar", ".rb", ".reg", ".scf", ".scr", ".sh", ".sys", ".tar", ".vbe", ".vbs", ".vhd", ".vhdx", ".wsf", ".wsh", ".zip", ".zsh"]);
let RiskyPaths = dynamic(["\\appdata\\", "\\mnt\\", "\\programdata\\", "\\public\\", "\\startup\\", "\\tmp\\", "\\temp\\", "\\volumes\\", "\\windows\\system32\\tasks\\"]);
// Find files with extensions that are commonly abused, written or modified by executables common to attack patterns.
let DroppedArtifacts = DeviceFileEvents
| where TimeGenerated between (lookback .. endwindow)
| where tolower(InitiatingProcessFileName) in (MicrosoftBins) and tolower(FileName) has_any (RiskyExtensions) and tolower(FolderPath) has_any (RiskyPaths)
| project DropTime = TimeGenerated, AccountName = InitiatingProcessAccountName, DeviceId, DeviceName, SHA1, FolderPath, DroppedFileName = tolower(FileName), DroppedBy = tolower(InitiatingProcessFileName), DroppedCommand = InitiatingProcessCommandLine;
// Find files executed.
let ProcessExecution = DeviceProcessEvents
| where TimeGenerated between (lookback .. endwindow)
| project ExecutionTime = TimeGenerated, AccountName, DeviceId, DeviceName, SHA1, ExecutedFileName = tolower(FileName), ProcessCommandLine, ExecutedBy = tolower(InitiatingProcessFileName), ExecutionType = "Process", ExecutedCommand = ProcessCommandLine;
// Find files loaded into memory.
let ImageLoadExecution = DeviceImageLoadEvents
| where TimeGenerated between (lookback .. endwindow)
| where tolower(FileName) has_any (".dll", ".ocx", ".sys")
| project ExecutionTime = TimeGenerated, AccountName = InitiatingProcessAccountName, DeviceId, DeviceName, SHA1, ExecutedFileName = tolower(FileName), ExecutedBy = tolower(InitiatingProcessFileName), ExecutedCommand = InitiatingProcessCommandLine, ExecutionType = "ImageLoad";
// Cross-reference files that were written or modified and then loaded or executed within 5 minutes.
let SubsequentExecution = union ProcessExecution, ImageLoadExecution;
DroppedArtifacts
| join kind = inner SubsequentExecution on DeviceId
| extend DropEndTime = DropTime + 5m, FileName = DroppedFileName
| where ExecutionTime between (DropTime .. DropEndTime) and DroppedFileName == ExecutedFileName and ExecutedBy != DroppedBy
// Filter normal execution paths and indeterminate noise.
and not(DroppedBy in ("powershell.exe", "powershell_ise.exe", "pwsh.exe") and FileName has_any (".dll", ".exe", ".ps1", ".psm1"))
| extend AccountName = tolower(AccountName)
// Group the events together by process chain.
| summarize events = count(), FolderPath = make_set(FolderPath), FileName = make_set(FileName), DropTime = make_set(DropTime), ExecutionTime = make_set(ExecutionTime), AccountName = make_set(AccountName), Switched = dcount(AccountName), DroppedCommand = make_set(DroppedCommand), ExecutedCmd = make_set(ExecutedCommand), ExecutedBy = make_set(ExecutedBy), ExecutionType = make_set(ExecutionType), SHA1 = make_set(SHA1) by DeviceName, DroppedBy
| where not(set_has_element(ExecutedBy, "ccmexec.exe"))
| extend Confidence = iff(set_has_element(AccountName, "system"), "Medium", iff(Switched > 1, "High", "Low"))
| project-away Switched
